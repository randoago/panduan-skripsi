\begin{isian}1.~~OBJEK/TEMPAT PENELITIAN\end{isian}
\kotak{%
\begin{itemize}
\item Tuliskan dengan jelas objek/tempat penelitian beserta alamatnya.
\item Khusus untuk penelitian yang tidak menggunakan objek, tempat penelitian dapat menggunakan alamat “Laboratorium Pemrograman/Laboratorium Robotika/Laboratorium Jaringan UM Buton”.
\end{itemize}}

\begin{isian}2.~~LATAR BELAKANG MASALAH\end{isian}
\kotak{Menjelaskan permasalahan yang dihadapi, diuraikan secara jelas, ringkas dan sistematis.}

\begin{isian}3.~~RUMUSAN MASALAH\end{isian}
\kotak{Berisi penjelasan tentang rumusan masalah yang telah disebutkan di bagian latar belakang, dituliskan dalam bentuk penomoran.}

\begin{isian}4.~~TUJUAN PENELITIAN\end{isian}
\kotak{Berisi penjelasan tentang tujuan penelitian (menjawab pernyataan dari bagian rumusan masalah).}

\begin{isian}5.~~PENELITIAN TERKAIT\end{isian}
\kotak{Cantumkan minimal 10 penelitian terkait/relevan dengan judul penelitian yang akan diajukan, dapat diambil dari jurnal/prosiding.

Buatlah kesimpulan dari masing-masing penelitian yang dirujuk serta tunjukkan perbedaan dan persamaan dengan penelitian yang dilakukan, mencakup permasalahan yang diteliti atau metode/algoritma yang digunakan.

Format penulisannya sebagai berikut:
\begin{enumerate}
\item Untuk jurnal/prosiding: tuliskan judul lengkap artikel ilmiah, nama pengarang, nama perguruan tinggi, nama jurnal/prosiding, Vol./Edisi, tahun terbit dan nomor ISSN.
\end{enumerate}
\textbf{Contoh:}

\textbf{“PERAN TEKNOLOGI INFORMASI DAN KOMUNIKASI DALAM PENDIDIKAN”; Haris Budiman; Universitas Islam Negeri Raden Intan Lampung; Al-Tadzkiyyah: Jurnal Pendidikan Islam; Volume 8 No. I 2017; E-ISSN: 2528-2476.}

\textbf{Persamaan: ………………………………}

\textbf{Perbedaan: ………………………………}}

\begin{isian}6.~~DAFTAR PUSTAKA\end{isian}
\kotak{Isi dari bagian ini adalah rujukan dari jurnal yang terkait dengan penelitian yang diusulkan.
